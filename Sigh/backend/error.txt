from test1_yunet.models import Person
import pickle, numpy as np

for p in Person.objects.all():
    if not p.face_encoding:
        print(p.name, "-> no encoding")
        continue
    e = pickle.loads(p.face_encoding)
    e = np.asarray(e, dtype=np.float32)
    print(f"{p.name}: dim={e.shape} norm={np.linalg.norm(e):.4f} "
          f"mean={e.mean():.4f} std={e.std():.4f} "
          f"min={e.min():.4f} max={e.max():.4f}")

Expect: dim=(128,) (SFace is 128-D), norm≈1.0, std in the ~0.05–0.15 range, values roughly symmetric around zero.

Red flags:

    dim is not 128 → you stored the wrong thing

    norm far from 1.0 → normalization was skipped or corrupted

    std near 0 → you stored a constant (broken alignment / thread race)

    std enormous → corrupted memory from the race
